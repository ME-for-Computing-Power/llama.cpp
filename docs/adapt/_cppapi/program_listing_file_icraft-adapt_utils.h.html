    template <class Type>
    void paramDataLayout(std::shared_ptr<Type[]> des, const Array<int64_t>& src_dim, const Array<int64_t>& des_dim,
        int64_t ct_index) {
        //param的软件到硬件排布的实现
        auto size = std::accumulate(src_dim.begin(), src_dim.end(), int64_t(1), std::multiplies());
        auto src = std::shared_ptr<Type[]>(new Type[size]);
        std::memcpy(src.get(), des.get(), sizeof(Type) * size);
        auto ct = des_dim[ct_index];
        auto cu = des_dim.back();
        //(pre,ct,middle,cu)
        auto middle = std::accumulate(des_dim.begin() + ct_index + 1, des_dim.end() - 1, int64_t(1), std::multiplies());
        auto pre = std::accumulate(des_dim.begin(), des_dim.begin() + ct_index, int64_t(1), std::multiplies());
        int64_t des_index = 0;
        for (int64_t i_pre = 0; i_pre < pre; i_pre++) {
            for (int64_t i_ct = 0; i_ct < ct; i_ct++) {
                for (int64_t i_middle = 0; i_middle < middle; i_middle++) {
                    for (int64_t i_cu = 0; i_cu < cu; i_cu++) {
                        int64_t src_index = i_pre * ct * middle * cu + i_middle * ct * cu + i_ct * cu + i_cu;
                        des[des_index] = src[src_index];
                        des_index++;
                    }
                }
            }
        }
    }

    template <class Type>
    void matmulWeightDataLayout(std::shared_ptr<Type[]> des, const Array<int64_t>& src_dim, const Array<int64_t>& des_dim) {
        auto size = std::accumulate(src_dim.begin(), src_dim.end(), int64_t(1), std::multiplies());
        auto src = std::shared_ptr<Type[]>(new Type[size]);
        std::memcpy(src.get(), des.get(), sizeof(Type) * size);
        auto Cit = des_dim[-3];
        auto Co = des_dim[-2];
        auto Ciu = des_dim[-1];
        auto Ci = Cit * Ciu;
        auto other = std::accumulate(des_dim.begin(), des_dim.end() - 3, int64_t(1), std::multiplies());
        int64_t des_index = 0;

        for (int64_t i_other = 0; i_other < other; i_other++) {
            for (int64_t i_Cit = 0; i_Cit < Cit; i_Cit++) {
                for (int64_t i_Co = 0; i_Co < Co; i_Co++) {
                    for (int64_t i_Ciu = 0; i_Ciu < Ciu; i_Ciu++) {
                        int64_t src_index = i_other * Co * Ci + (i_Cit * Ciu + i_Ciu) * Co + i_Co;
                        des[des_index] = src[src_index];
                        des_index++;
                    }
                }
            }
        }
    }

    template <class Type>
    std::shared_ptr<Type[]> depthwiseWeightDataMakeUp(std::shared_ptr<Type[]> src, const Array<int64_t>& des_dim,
        const MergedAxisDistr& ichan_distr) {
        auto size = std::accumulate(des_dim.begin(), des_dim.end(), int64_t(1), std::multiplies());
        auto des = std::shared_ptr<Type[]>(new Type[size]);
        memset(des.get(), 0, sizeof(Type) * size);

        auto ofm_c = des_dim[3];
        auto kernel_size = des_dim[0] * des_dim[1];
        int64_t src_index = 0;
        for (int64_t i_kernel = 0; i_kernel < kernel_size; i_kernel++) {
            for (int64_t i_co = 0; i_co < ofm_c; i_co++) {
                if (ichan_distr.isValid(i_co)) {
                    int64_t des_index = i_kernel * ofm_c + i_co;
                    if (des_index >= size) throw(Error("index error in depthwiseWeightDataMakeUp"));
                    des[des_index] = src[src_index];
                    src_index++;
                }
            }
        }
        return des;
    }

    template <class Type>
    std::shared_ptr<Type[]> splitWeightDataMakeUp(Type value, const std::vector<int64_t>& dim, const MergedAxisDistr& ochan_distr) {
        auto dim_size = dim.size();
        int kernel_ic = dim[dim_size - 2];
        int kernel_oc = dim[dim_size - 1];
        auto size = std::accumulate(dim.begin(), dim.end(), (int64_t)1, std::multiplies());
        auto data = std::shared_ptr<Type[]>(new Type[size]);
        std::memset(data.get(), 0, size * sizeof(Type));

        int ko_valid_offset = 0;
        for (auto&& [s, e] : ochan_distr.validSections()) {
            int chan_num = e - s;
            for (auto k_i = 0; k_i < kernel_ic; k_i++) {
                for (auto k_o = 0; k_o < chan_num; k_o++) {
                    auto ko_valid = ko_valid_offset + k_o;
                    if (ko_valid == k_i) {
                        int index = k_i * kernel_oc + s + k_o;
                        data[index] = value;
                    }
                }
            }
            ko_valid_offset += chan_num;
        }
        return data;
    }
