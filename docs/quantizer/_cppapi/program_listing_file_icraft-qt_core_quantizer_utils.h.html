        template <typename T1, typename T2>
        static void transposenio2oin(const T1* src, T2* des, const std::vector<int64_t> dim) {
            if (dim.size() != 3) {
                throw("weights error\n");
            }
            else {
                int index_src;
                int index_des;
                int kn = dim[0];
                int ki = dim[1];
                int ko = dim[2];
                for (int in = 0; in < kn; in++) {
                    for (int ii = 0; ii < ki; ii++) {
                        for (int io = 0; io < ko; io++) {
                            index_src = in * ki * ko + ii * ko + io;
                            index_des = io * ki * kn + ii * kn + in;
                            des[index_des] = src[index_src];
                        }
                    }
                }
            }
        }

        static int fileSize(std::ifstream& file_stream) {
            file_stream.seekg(0, file_stream.end);
            auto file_size = file_stream.tellg();
            file_stream.seekg(0, file_stream.beg);
            return file_size;
        }

        static bool fileExists(const std::string& filename) {
            struct stat buf;
            if (stat(filename.data(), &buf) != -1) {
                return true;
            }
            return false;
        }

        static void* loadDLLByPath(const std::string& path) {
            if (fileExists(path)) {
#ifdef _WIN32
                auto h = LoadLibraryA(path.c_str());
                if (h == NULL) {
                    std::cerr << "load dll " << path << " error: " << std::to_string(GetLastError()) << std::endl;
                    //throw Error("Load DLL \"" + path + "\" error: " + std::to_string(GetLastError()));
                }
                return h;
#else
                auto h = dlopen(path.data(), RTLD_LAZY);
                if (h == NULL) {
                    throw Error("Load SO \"" + path + "\" error: " + dlerror());
                }
                return h;

#endif
            }
            //else throw FileNotExistError(path);
            else {
                std::cerr << path << " not exit error!" << std::endl;
                exit(-1);
            }
        }

        static QuantizerAdapt* getQuantizer(void* h) {
#ifdef _WIN32
            if (h == NULL) {
                ICRAFT_LOG(EXCEPT, 21010).append("DLL is not loaded!");
            }
            else {
                auto getQuantizer = (GetDll)GetProcAddress((HINSTANCE)h, "GetQuantizer");
                if (getQuantizer != NULL) {
                    return getQuantizer();
                }
                else {
                    ICRAFT_LOG(EXCEPT, 21010).append("Get process 'GetQuantizer' error: {}", GetLastError());
                    return nullptr;
                }
            }
#else
            if (h == NULL) {
                throw Error("SO is not loaded!");
            }
            else {
                dlerror();
                auto getUserCustom = (GetDll)dlsym(h, "GetQuantizer");
                char* error;
                if ((error = dlerror()) == NULL) {
                    return getUserCustom();
                }
                else {
                    ICRAFT_LOG(EXCEPT, 21010).append("Get process 'GetQuantizer' error: {}", error);
                    return nullptr;
                }
            }
#endif
        }
